// Appended through an overlay to enrollment_test.go; imports/helpers are existing.
func TestResearchExistingProtectedInventory(t *testing.T) {
 root, request := enrolledFixture(t)
 files := map[string][]byte{}
 entries, err := os.ReadDir(root)
 if err != nil { t.Fatal(err) }
 for _, entry := range entries {
  if entry.Name() == manifestName { continue }
  body, err := os.ReadFile(filepath.Join(root, entry.Name()))
  if err != nil { t.Fatal(err) }
  files[entry.Name()] = body
 }
 repin := func() {
  t.Helper()
  manifest := makeManifest(t, files)
  if err := os.WriteFile(filepath.Join(root, manifestName), manifest, 0600); err != nil { t.Fatal(err) }
  hash := sha256.Sum256(manifest)
  request.Pin.ManifestSHA256 = hex.EncodeToString(hash[:])
 }
 add := func(name string) {
  t.Helper()
  files[name] = []byte("synthetic resource " + name)
  if err := os.WriteFile(filepath.Join(root, name), files[name], 0600); err != nil { t.Fatal(err) }
 }
 add("ardents-node-linux-amd64"); add("ardents-custody-linux-amd64"); repin()
 if _, err := VerifyHeadless(request); err != nil { t.Fatal(err) }
 t.Log("old v3 headless without text resources: accepted (not protected installation)")
 resources := []string{"ardents-text-linux-amd64", "ardents-text-reader@.service", "ardents-text-publisher@.service", "ardents-text-reader.socket", "ardents-text-publisher.socket", "50-ardents-text.rules", "ardents-text.conf", "ardents-endpoint.service"}
 oldPin := request.Pin.ManifestSHA256
 for _, name := range resources { add(name) }; repin()
 newPin := request.Pin.ManifestSHA256
 request.Pin.ManifestSHA256 = oldPin
 if _, err := VerifyHeadless(request); err == nil { t.Fatal("changed manifest accepted with predecessor pin") }
 t.Log("changed inventory with predecessor pin: refused")
 request.Pin.ManifestSHA256 = newPin
 verified, err := VerifyHeadless(request)
 if err != nil { t.Fatal(err) }
 for _, name := range resources {
  if !bytes.Equal(verified.Inputs.Files[release.MetadataURL(name)], files[name]) { t.Fatalf("%s did not enter metadata", name) }
 }
 t.Log("complete synthetic text inventory: authenticated but all eight resources enter Release metadata")
 partial := resources[1]
 if err := os.Remove(filepath.Join(root, partial)); err != nil { t.Fatal(err) }
 delete(files, partial); repin()
 if _, err := VerifyHeadless(request); err != nil { t.Fatalf("partial text set unexpectedly refused: %v", err) }
 t.Log("re-pinned partial text inventory: accepted")
 add(partial)
 if _, err := VerifyHeadless(request); err == nil { t.Fatal("undeclared extra file accepted") }
 t.Log("undeclared extra file: refused")
 repin()
 if err := os.WriteFile(filepath.Join(root, resources[0]), []byte("substituted worker"), 0600); err != nil { t.Fatal(err) }
 if _, err := VerifyHeadless(request); err == nil { t.Fatal("worker digest substitution accepted") }
 t.Log("worker digest substitution: refused")
}
